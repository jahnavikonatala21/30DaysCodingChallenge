class Solution {
    public int[] smallerNumbersThanCurrent(int[] nums) {
        HashMap<Integer,Integer> mp=new HashMap<>();
        int[] arr=nums.clone();
        Arrays.sort(arr);
        for(int i=0;i<arr.length;i++){
            if(!mp.containsKey(arr[i])){
                mp.put(arr[i],i);
            }
        }
        int[] res=new int[nums.length];
        for(int i=0;i<nums.length;i++){
            res[i]=(mp.get(nums[i]));
        }
        return res;
    }
}
