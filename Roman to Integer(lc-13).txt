class Solution {
    public int romanToInt(String s) {
        int[] map=new int[128]; //ASCII Level
        map['I']=1;
        map['V']=5;
        map['X']=10;
        map['L']=50;
        map['C']=100;
        map['D']=500;
        map['M']=1000;

        int sum=0,prev=0;
        for (int i=s.length()-1;i>=0;i--){
            int val=map[s.charAt(i)];
            if(val<prev){
                sum-=val;
            }else{
                sum+=val;
            }
            prev=val;
        }
        return sum;
    }
}
