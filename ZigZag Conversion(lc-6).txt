class Solution {
    public String convert(String s, int numRows) {
        if(numRows==1) return s;
        StringBuilder res=new StringBuilder();
        int len=s.length();
        for(int r=0;r<numRows;r++){
            int incr=2*(numRows-1);
            for(int i=r;i<len;i+=incr){
                res.append(s.charAt(i)); //for all elements except diagonals
                if(r>0 && r<numRows-1 && i+incr-2*r<len){
                    res.append(s.charAt(i+incr-2*r));
                }
            }
        }
        return res.toString();
        
    }
}
